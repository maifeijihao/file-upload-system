#!/bin/bash
cd /opt/upload

case "$1" in
    start)
        echo "启动服务..."
        python3 server.py
        ;;
    stop)
        echo "停止服务..."
        pkill -f "python3 server.py"
        sleep 2
        ;;
    restart)
        echo "重启服务..."
        pkill -f "python3 server.py" 2>/dev/null
        sleep 2
        python3 server.py
        ;;
    status)
        if pgrep -f "python3 server.py" > /dev/null; then
            echo "✅ 服务运行中"
            PORT=$(ps aux | grep "python3 server.py" | grep -v grep | grep -o "port=[0-9]*" | cut -d'=' -f2)
            IP=$(hostname -I | awk '{print $1}')
            echo "访问地址: http://$IP:${PORT:-5000}"
        else
            echo "❌ 服务未运行"
        fi
        ;;
    *)
        echo "用法: $0 {start|stop|restart|status}"
        ;;
esac
